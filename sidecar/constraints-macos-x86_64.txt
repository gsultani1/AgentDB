# Legacy ML stack for Intel-Mac (darwin-x64) sidecar builds.
#
# PyTorch stopped publishing macOS x86_64 wheels after the 2.2 series
# (deprecated January 2024), so torch 2.2.2 is the newest Intel-Mac torch
# that will ever exist. Everything here is the last mutually compatible
# generation around it: torch 2.2 was compiled against NumPy 1.x (NumPy 2
# crashes it), and transformers >= 4.42-era releases require torch >= 2.4.
#
# Used only by .github/workflows/sidecar.yml for the macos-15-intel job:
#   python -m pip install -e . ... -c sidecar/constraints-macos-x86_64.txt
# All other platforms build with the current unconstrained stack.
numpy==1.26.4
torch==2.2.2
transformers==4.41.2
sentence-transformers==3.0.1
